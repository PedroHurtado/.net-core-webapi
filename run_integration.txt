# Ejecutar tests de integración
dotnet test tests/WebApi.IntegrationTests

# Ejecutar con más detalle (verbosity normal)
dotnet test tests/WebApi.IntegrationTests --verbosity normal

# Ejecutar con cobertura de código
dotnet test tests/WebApi.IntegrationTests --collect:"XPlat Code Coverage"

# Ejecutar solo tests de integración con filtro
dotnet test tests/WebApi.IntegrationTests --filter "FullyQualifiedName~CreateIngredientTests"

# Generar reporte HTML de cobertura (requiere ReportGenerator instalado)
reportgenerator -reports:"tests/WebApi.IntegrationTests/TestResults/*/coverage.cobertura.xml" -targetdir:"coveragereport-integration" -reporttypes:Html

# Abrir reporte de cobertura (Linux/Mac)
open coveragereport-integration/index.html

# Abrir reporte de cobertura (Windows)
start coveragereport-integration/index.html

# Ejecutar todos los tests (unitarios + integración)
dotnet test

# Ejecutar todos los tests con cobertura
dotnet test --collect:"XPlat Code Coverage"

# Comando todo en uno: test + reporte + abrir (Linux/Mac)
dotnet test tests/WebApi.IntegrationTests --collect:"XPlat Code Coverage" && reportgenerator -reports:"tests/WebApi.IntegrationTests/TestResults/*/coverage.cobertura.xml" -targetdir:"coveragereport-integration" -reporttypes:Html && start coveragereport-integration/index.html
