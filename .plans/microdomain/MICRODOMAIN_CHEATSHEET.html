⚡ Fase 2: Commands (Paralelo - TDD)

① Test red → ② Command green → ③ PR
// Domain/[Aggregate]/Commands/[Aggregate]_Update.cs public partial class Menu { [Injectable(ServiceLifetime.Singleton)] public class Update(IValidator<Menu> validator) : AbstractModifyCommand<UpdateMenuCommand, Menu> { public override Menu Handle( Menu entity, UpdateMenuCommand cmd) { entity.Name = cmd.Name; return validator.ValidateOrThrow(entity); } } }
