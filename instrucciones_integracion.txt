# 1. Crear el proyecto de tests de integración
dotnet new xunit -n WebApi.IntegrationTests -o tests/WebApi.IntegrationTests

# 2. Agregar el proyecto a la solución (si tienes un .sln)
dotnet sln add tests/WebApi.IntegrationTests/WebApi.IntegrationTests.csproj

# 3. Agregar referencia al proyecto principal
dotnet add tests/WebApi.IntegrationTests reference src/webapi/webapi.csproj

# 4. Instalar paquetes necesarios para tests de integración
dotnet add tests/WebApi.IntegrationTests package Microsoft.AspNetCore.Mvc.Testing
dotnet add tests/WebApi.IntegrationTests package FluentAssertions
dotnet add tests/WebApi.IntegrationTests package coverlet.collector
dotnet add tests/WebApi.IntegrationTests package Microsoft.EntityFrameworkCore.InMemory

# 5. Crear estructura de carpetas
mkdir -p tests/WebApi.IntegrationTests/Features/Ingredients

# 6. Ejecutar los tests de integración
dotnet test tests/WebApi.IntegrationTests

# 7. Ejecutar tests de integración con cobertura
dotnet test tests/WebApi.IntegrationTests --collect:"XPlat Code Coverage"

# 8. Generar reporte de cobertura (requiere ReportGenerator instalado)
reportgenerator -reports:"tests/WebApi.IntegrationTests/TestResults/*/coverage.cobertura.xml" -targetdir:"coveragereport-integration" -reporttypes:Html

# 9. Abrir reporte (Linux/Mac)
open coveragereport-integration/index.html

# 10. Ejecutar todos los tests (unitarios e integración)
dotnet test

# 11. Ejecutar todos los tests con cobertura combinada
dotnet test --collect:"XPlat Code Coverage"
